\text{Predictor: } \tilde{w}_{i+1} &= w_i + h f(t_i, w_i) \\
\text{Corrector: } w_{i+1} &= w_i + \frac{h}{2} \left[ f(t_i, w_i) + f(t_{i+1}, \tilde{w}_{i+1}) \right]
